package org.apache.tika.detect;

import org.apache.tika.detect.ZeroSizeFileDetector;
import org.apache.tika.io.TikaInputStream;
import org.apache.tika.metadata.Metadata;
import org.apache.tika.parser.ParseContext;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import org.apache.tika.mime.MediaType;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class ZeroSizeFileDetector_detect_0_2_Test {

    @Mock
    private ParseContext parseContext;

    private ZeroSizeFileDetector detector;

    @BeforeEach
    public void setUp() {
        MockitoAnnotations.openMocks(this);
        detector = new ZeroSizeFileDetector();
    }

    @Test
    public void testDetectNormalOperation() throws IOException {
        TikaInputStream tis = createTikaInputStream("test.txt", 1024);
        Metadata metadata = new Metadata();
        MediaType expectedMediaType = MediaType.OCTET_STREAM;
        MediaType actualMediaType = detector.detect(tis, metadata, parseContext);
        assertEquals(expectedMediaType, actualMediaType);
    }

    @Test
    public void testDetectNullInputStream() throws IOException {
        Metadata metadata = new Metadata();
        ParseContext parseContext = mock(ParseContext.class);
        MediaType expectedMediaType = MediaType.OCTET_STREAM;
        MediaType actualMediaType = detector.detect(null, metadata, parseContext);
        assertEquals(expectedMediaType, actualMediaType);
    }

    @Test
    public void testDetectZeroLengthInputStream() throws IOException {
        TikaInputStream tis = createTikaInputStream("test.txt", 0);
        Metadata metadata = new Metadata();
        ParseContext parseContext = mock(ParseContext.class);
        MediaType expectedMediaType = MediaType.EMPTY;
        MediaType actualMediaType = detector.detect(tis, metadata, parseContext);
        assertEquals(expectedMediaType, actualMediaType);
    }

    @Test
    public void testDetectNullMetadata() throws IOException {
        TikaInputStream tis = createTikaInputStream("test.txt", 1024);
        ParseContext parseContext = mock(ParseContext.class);
        MediaType expectedMediaType = MediaType.OCTET_STREAM;
        MediaType actualMediaType = detector.detect(tis, null, parseContext);
        assertEquals(expectedMediaType, actualMediaType);
    }

    @Test
    public void testDetectNullParseContext() throws IOException {
        TikaInputStream tis = createTikaInputStream("test.txt", 1024);
        Metadata metadata = new Metadata();
        MediaType expectedMediaType = MediaType.OCTET_STREAM;
        MediaType actualMediaType = detector.detect(tis, metadata, null);
        assertEquals(expectedMediaType, actualMediaType);
    }

    @Test
    public void testDetectEmptyInputStream() throws IOException {
        TikaInputStream tis = createTikaInputStream("test.txt", 0);
        Metadata metadata = new Metadata();
        ParseContext parseContext = mock(ParseContext.class);
        MediaType expectedMediaType = MediaType.EMPTY;
        MediaType actualMediaType = detector.detect(tis, metadata, parseContext);
        assertEquals(expectedMediaType, actualMediaType);
    }

    @Test
    public void testDetectLargeInputStream() throws IOException {
        TikaInputStream tis = createTikaInputStream("test.txt", 1024 * 1024);
        Metadata metadata = new Metadata();
        ParseContext parseContext = mock(ParseContext.class);
        MediaType expectedMediaType = MediaType.OCTET_STREAM;
        MediaType actualMediaType = detector.detect(tis, metadata, parseContext);
        assertEquals(expectedMediaType, actualMediaType);
    }

    private TikaInputStream createTikaInputStream(String fileName, long length) throws IOException {
        Path path = Paths.get(fileName);
        try (InputStream inputStream = Files.newInputStream(path)) {
            return new TikaInputStream(inputStream, length);
        }
    }
}
