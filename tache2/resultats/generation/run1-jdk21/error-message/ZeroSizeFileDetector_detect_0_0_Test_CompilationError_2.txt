public class ZeroSizeFileDetector_detect_0_0_Test {

    @Test
    public void testDetectWithNonEmptyStream() throws IOException {
        // Create a TikaInputStream with a non-empty byte array
        TikaInputStream tis = TikaInputStream.get(new ByteArrayInputStream("Hello, World!".getBytes()));
        // Create a Metadata object
        Metadata metadata = new Metadata();
        // Create a ParseContext object
        ParseContext parseContext = new ParseContext();
        // Call the detect method
        MediaType result = ZeroSizeFileDetector.detect(tis, metadata, parseContext);
        // Assert the result
        assertEquals(MediaType.TEXT_PLAIN, result);
    }

    @Test
    public void testDetectWithEmptyStream() throws IOException {
        // Create a TikaInputStream with an empty byte array
        TikaInputStream tis = TikaInputStream.get(new ByteArrayInputStream(new byte[0]));
        // Create a Metadata object
        Metadata metadata = new Metadata();
        // Create a ParseContext object
        ParseContext parseContext = new ParseContext();
        // Call the detect method
        MediaType result = ZeroSizeFileDetector.detect(tis, metadata, parseContext);
        // Assert the result
        assertEquals(MediaType.EMPTY, result);
    }
