package org.apache.tika.detect;

import org.apache.tika.detect.ZeroSizeFileDetector;
import org.apache.tika.io.TikaInputStream;
import org.apache.tika.metadata.Metadata;
import org.apache.tika.mime.MediaType;
import org.apache.tika.parser.ParseContext;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.ValueSource;
import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.io.InputStream;
import java.lang.reflect.Constructor;
import java.lang.reflect.InvocationTargetException;
import java.util.Properties;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import org.apache.tika.annotation.TikaComponentProcessor;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class ZeroSizeFileDetector_detect_0_1_Test {

    private ZeroSizeFileDetector detector;

    @BeforeEach
    public void setUp() {
        detector = new ZeroSizeFileDetector();
    }

    @Test
    public void testDetectWithNullTikaInputStream() throws IOException {
        MediaType result = detector.detect(null, new Metadata(), new ParseContext());
        assertEquals(MediaType.OCTET_STREAM, result);
    }

    @Test
    public void testDetectWithEmptyTikaInputStream() throws IOException {
        TikaInputStream tis = new TikaInputStream(new ByteArrayInputStream(new byte[0]), 0);
        MediaType result = detector.detect(tis, new Metadata(), new ParseContext());
        assertEquals(MediaType.EMPTY, result);
    }

    @Test
    public void testDetectWithNonEmptyTikaInputStream() throws IOException {
        TikaInputStream tis = new TikaInputStream(new ByteArrayInputStream(new byte[1]), 1);
        MediaType result = detector.detect(tis, new Metadata(), new ParseContext());
        assertEquals(MediaType.OCTET_STREAM, result);
    }

    @ParameterizedTest
    @ValueSource(booleans = { true, false })
    public void testDetectWithMarkAndReset(boolean markSupported) throws IOException {
        TikaInputStream tis = new TikaInputStream(new ByteArrayInputStream(new byte[1]), 1);
        if (markSupported) {
            tis.mark(1);
            tis.reset();
        }
        MediaType result = detector.detect(tis, new Metadata(), new ParseContext());
        assertEquals(MediaType.OCTET_STREAM, result);
    }

    @Test
    public void testDetectWithIOException() throws IOException {
        TikaInputStream tis = mock(TikaInputStream.class);
        when(tis.read()).thenThrow(new IOException("Simulated I/O error"));
        MediaType result = detector.detect(tis, new Metadata(), new ParseContext());
        assertEquals(MediaType.OCTET_STREAM, result);
    }

    @Test
    public void testDetectWithInvalidMetadata() throws IOException {
        TikaInputStream tis = new TikaInputStream(new ByteArrayInputStream(new byte[1]), 1);
        Metadata metadata = null;
        MediaType result = detector.detect(tis, metadata, new ParseContext());
        assertEquals(MediaType.OCTET_STREAM, result);
    }

    @Test
    public void testDetectWithInvalidParseContext() throws IOException {
        TikaInputStream tis = new TikaInputStream(new ByteArrayInputStream(new byte[1]), 1);
        ParseContext parseContext = null;
        MediaType result = detector.detect(tis, new Metadata(), parseContext);
        assertEquals(MediaType.OCTET_STREAM, result);
    }

    @Test
    public void testDetectWithInvalidInputSource() throws IOException {
        TikaInputSource inputSource = new TikaInputSource(null);
        TikaInputStream tis = new TikaInputStream(inputSource, mock(TemporaryResources.class), null);
        MediaType result = detector.detect(tis, new Metadata(), new ParseContext());
        assertEquals(MediaType.OCTET_STREAM, result);
    }
}
